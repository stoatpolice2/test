【コードから手動作成する手順】
  1. パソコン上に新規フォルダ（例：my-novel-game）を作成します。
  2. 下記の各タブから「コードをコピー」ボタンを押し、指定のファイル名で保存してください。
  3. すべてのファイルを同じフォルダ内に配置すれば、ツールのセットアップ完了です！

  async function loadScenarioFromTxt(filePath) {
    try {
      const response = await fetch(filePath);
      const text = await response.text();
      const lines = text.split(/\r?\n/).filter(line => line.trim() !== "");
      const parsedScenario = []; const labelMap = {}; let currentScene = {}; let sceneIdCounter = 0;

      lines.forEach(line => {
        line = line.trim();
        if (line.startsWith("@")) {
          const firstColonIdx = line.indexOf(":");
          let command = ""; let param = "";
          if (firstColonIdx !== -1) { command = line.slice(1, firstColonIdx).trim(); param = line.slice(firstColonIdx + 1).trim(); }
          else { command = line.slice(1).trim(); }

          if (command === "label") { labelMap[param] = sceneIdCounter; currentScene.label = param; }
          else if (command === "bg") currentScene.bg = param;
          else if (command === "bgm") currentScene.bgm = param;
          else if (command === "character") {
            if (param.includes("|")) {
              const parts = param.split("|").map(s => s.trim());
              currentScene.character = parts[0];
              parts.slice(1).forEach(p => { if (p.startsWith("pos:")) currentScene.characterPos = p.replace("pos:", "").trim(); });
            } else { currentScene.character = param; currentScene.characterPos = "center"; }
          }
          else if (command === "fade") currentScene.fade = (param === "true");
          else if (command === "startCredits") currentScene.startCredits = (param === "true");
          else if (command === "branch") {
            currentScene.isBranch = true;
            const parts = param.split("|");
            parts.slice(1).forEach(p => {
              if (p.startsWith("happy:")) currentScene.happyTarget = p.replace("happy:", "").trim();
              if (p.startsWith("normal:")) currentScene.normalTarget = p.replace("normal:", "").trim();
            });
          }
          else if (command === "choice") {
            const choiceItems = param.split("/");
            currentScene.choices = choiceItems.map(item => {
              const parts = item.split("|").map(s => s.trim());
              const text = parts[0]; let nextTarget = null; let favorChange = 0;
              parts.slice(1).forEach(p => {
                if (p.startsWith("next:")) nextTarget = p.replace("next:", "").trim();
                if (p.startsWith("favor:")) favorChange = parseInt(p.replace("favor:", ""));
              });
              return { text, nextTarget, favorChange };
            });
          }
          return;
        }

        if (line.includes(":")) {
          const [speaker, ...textParts] = line.split(":");
          currentScene.speaker = speaker.trim(); currentScene.text = textParts.join(":").trim();
        } else { currentScene.speaker = ""; currentScene.text = line; }

        currentScene.id = sceneIdCounter; currentScene.next = sceneIdCounter + 1;
        parsedScenario.push(currentScene); sceneIdCounter++; currentScene = {};
      });

      parsedScenario.forEach(scene => {
        if (scene.choices) {
          scene.choices.forEach(choice => {
            if (choice.nextTarget) {
              choice.next = (labelMap[choice.nextTarget] !== undefined) ? labelMap[choice.nextTarget] : parseInt(choice.nextTarget);
            }
          });
        }
        if (scene.happyTarget && labelMap[scene.happyTarget] !== undefined) scene.happyNext = labelMap[scene.happyTarget];
        if (scene.normalTarget && labelMap[scene.normalTarget] !== undefined) scene.normalNext = labelMap[scene.normalTarget];
      });

      return parsedScenario;
    } catch (error) { console.error("scenario.txt 読み込みエラー:", error); return []; }
  }

  window.updateScenarioFromText = function(txtContent) {
    const lines = txtContent.split(/\r?\n/).filter(line => line.trim() !== "");
    const parsedScenario = []; const labelMap = {}; let currentScene = {}; let sceneIdCounter = 0;
    lines.forEach(line => {
      line = line.trim();
      if (line.startsWith("@")) {
        const firstColonIdx = line.indexOf(":"); let command = ""; let param = "";
        if (firstColonIdx !== -1) { command = line.slice(1, firstColonIdx).trim(); param = line.slice(firstColonIdx + 1).trim(); }
        else { command = line.slice(1).trim(); }
        if (command === "label") { labelMap[param] = sceneIdCounter; currentScene.label = param; }
        else if (command === "bg") currentScene.bg = param;
        else if (command === "bgm") currentScene.bgm = param;
        else if (command === "character") {
          if (param.includes("|")) {
            const parts = param.split("|").map(s => s.trim()); currentScene.character = parts[0];
            parts.slice(1).forEach(p => { if (p.startsWith("pos:")) currentScene.characterPos = p.replace("pos:", "").trim(); });
          } else { currentScene.character = param; currentScene.characterPos = "center"; }
        }
        else if (command === "fade") currentScene.fade = (param === "true");
        else if (command === "startCredits") currentScene.startCredits = (param === "true");
        else if (command === "branch") {
          currentScene.isBranch = true; const parts = param.split("|");
          parts.slice(1).forEach(p => { if (p.startsWith("happy:")) currentScene.happyTarget = p.replace("happy:", "").trim(); if (p.startsWith("normal:")) currentScene.normalTarget = p.replace("normal:", "").trim(); });
        }
        else if (command === "choice") {
          const choiceItems = param.split("/");
          currentScene.choices = choiceItems.map(item => {
            const parts = item.split("|").map(s => s.trim()); const text = parts[0]; let nextTarget = null; let favorChange = 0;
            parts.slice(1).forEach(p => { if (p.startsWith("next:")) nextTarget = p.replace("next:", "").trim(); if (p.startsWith("favor:")) favorChange = parseInt(p.replace("favor:", "")); });
            return { text, nextTarget, favorChange };
          });
        }
        return;
      }
      if (line.includes(":")) { const [speaker, ...textParts] = line.split(":"); currentScene.speaker = speaker.trim(); currentScene.text = textParts.join(":").trim(); }
      else { currentScene.speaker = ""; currentScene.text = line; }
      currentScene.id = sceneIdCounter; currentScene.next = sceneIdCounter + 1; parsedScenario.push(currentScene); sceneIdCounter++; currentScene = {};
    });
    parsedScenario.forEach(scene => {
      if (scene.choices) { scene.choices.forEach(choice => { if (choice.nextTarget) { choice.next = (labelMap[choice.nextTarget] !== undefined) ? labelMap[choice.nextTarget] : parseInt(choice.nextTarget); } }); }
      if (scene.happyTarget && labelMap[scene.happyTarget] !== undefined) scene.happyNext = labelMap[scene.happyTarget];
      if (scene.normalTarget && labelMap[scene.normalTarget] !== undefined) scene.normalNext = labelMap[scene.normalTarget];
    });
    scenario = parsedScenario; favorability = 0; backlog = []; stopAutoSkip(); showGameUI(); renderScene(0);
  };

Readme.txt 詳しい取扱説明書テキスト（メモ帳等で閲覧可能） 
========================================================================
【手軽に作れる！Webノベルゲーム制作ツール】取扱説明書 & シナリオ記述マニュアル
========================================================================

この度は本ツールをダウンロードいただき、ありがとうございます！
本ツールは、プログラミングの知識がなくても「scenario.txt」を書き換えるだけで、
ブラウザで遊べる本格的なWebノベルゲームを作成できる軽量ゲームエンジンです。


------------------------------------------------------------------------
■ 1. 収録ファイルとフォルダ構成
------------------------------------------------------------------------

・index.html    : ゲームのシステム本体（プレイヤーが遊ぶ画面）
・editor.html   : リアルタイムプレビュー付きシナリオエディタ（制作・編集用）
・scenario.txt  : ゲームのシナリオファイル（テキスト編集で作品を作ります）
・Readme.txt    : この説明書ファイル


------------------------------------------------------------------------
■ 2. 制作・編集方法（専用エディタ editor.html の使い方）
------------------------------------------------------------------------

本ツールには、ブラウザ上で直感的にシナリオを作成できる「専用エディタ」が付属しています。

【エディタの起動方法】
「editor.html」をブラウザ（Chrome, Edge, Safari等）で開いてください。
※ PhoenixCode や VS Code のプレビュー機能、またはローカルサーバー環境で開くと
  よりスムーズに動作します。

【エディタの特長・機能】
・画面左側（テキストエディタ）:
  構文強調（シンタックスハイライト）機能付きです。
  @コマンド、話者名（名前:）、コメント行などが自動で色分け表示されます。

・画面右側（リアルタイムプレビュー）:
  左側でシナリオを書き換えると、右側のゲーム画面へ即座に反映されます。
  実際にゲームを動かしながら演出やセリフのテンポを確認できます。

・「ファイルを開く」ボタン:
  すでに作成したシナリオファイル（scenario.txt）を読み込んで再編集できます。

・「scenario.txt を保存」ボタン:
  編集が完了したら、このボタンを押すだけでゲームにそのまま読み込める
  「scenario.txt」がダウンロード保存されます。


------------------------------------------------------------------------
■ 3. 動作確認・遊ぶ方法（プレイヤー環境）
------------------------------------------------------------------------

【作品を公開・配布する場合】
「index.html」と「scenario.txt」（および使用する画像・音声素材）を
各種Webサーバー（GitHub Pages、Netlify、レンタルサーバー等）にアップロードすれば、
スマホやPCのブラウザで誰でも遊べる状態で公開できます。


------------------------------------------------------------------------
■ 4. シナリオファイル（scenario.txt）の書き方ルール
------------------------------------------------------------------------

scenario.txt には「会話文」と「演出用コマンド（@から始まる行）」を記述します。

【基本のルール】
・話者名とセリフはコロン「:」で区切ります。
  （例） ほのか:こんにちは！
・「:」がない行は、話者名なしの「地の文（ト書き）」になります。
  （例） 目が覚めると、見知らぬ部屋にいた……。
・「@」から始まる行は、背景変更やBGM再生などのシステム命令です。
・指定（@bg や @character など）を省略した行は、直前の状態がそのまま維持されます。
・「#」または「//」で始まる行はコメント（メモ書き）となり、ゲームには表示されません。


------------------------------------------------------------------------
■ 5. 演出コマンド一覧
------------------------------------------------------------------------

◆ 背景画像の変更
   @bg: 画像URL
   （例） @bg: https://example.com/bg01.jpg
   ※空文字「@bg: 」にすると背景を消去（黒画面）します。

◆ BGM（音源）の再生・停止
   @bgm: 音楽ファイルURL
   （例） @bgm: https://example.com/bgm01.ogg
   ※空文字「@bgm: 」にするとBGMを停止します。

◆ 立ち絵画像の表示・位置指定
   @character: 画像URL|pos:位置
   （例） @character: https://example.com/chara01.png|pos:left
   ※位置（pos）は left（左）、center（中央）、right（右）から指定できます。
   ※省略時は center になります。
   ※空文字「@character: 」にすると立ち絵を消去（非表示）します。

◆ 画面フェード（暗転）切り替え
   @fade: true
   ※直後のセリフに移動する際、黒画面を介して滑らかに切り替えます。

◆ 移動先（ジャンプ先）のラベル作成
   @label: ラベル名
   （例） @label: happy_end
   ※選択肢の飛び先やルート分岐先の「目印」として配置します。

◆ 選択肢の表示と好感度加算
   @choice: ボタン文章|next:ラベル名|favor:加算値 / ボタン文章2|next:ラベル名2|favor:加算値
   （例） @choice: 一緒に行く|next:route_a|favor:10 / 断る|next:route_b|favor:-10
   ※選択肢ボタンを画面に表示します。「/」で区切ることで複数の選択肢を作れます。
   ※「favor:数値」で、プレイヤーの隠しパラメータ（好感度）を加減算します。

◆ 好感度によるルート自動分岐
   @branch: true|happy:HAPPY側ラベル|normal:NORMAL側ラベル
   （例） @branch: true|happy:happy_end|normal:normal_end
   ※好感度が「10以上」なら happy のラベルへ、「10未満」なら normal のラベルへ自動ジャンプします。

◆ スタッフロールの開始
   @startCredits: true
   ※直後のセリフを読み終えた後、タイトルBGMを継続させたままスタッフロール画面へ移動します。

■ 6. プレイヤーが使えるシステム機能
------------------------------------------------------------------------

ゲームのプレイ中、画面右上（スマホは上部）のメニューから以下の機能が使えます。

・AUTO  : 文章を自動で読み進めます。（AキーでもON/OFF可能）
・SKIP  : 一度読んだシーン（既読文章）のみを高速で飛ばします。（SキーでもON/OFF可能）
・LOG   : これまでの会話履歴（バックログ）を確認します。画面の上スワイプや上ホイールでも開けます。
・SAVE  : 最大3つのスロットにゲームの進行状況を保存します。
・LOAD  : セーブしたデータを読み込んで再開します。
・TITLE : タイトル画面に戻ります。
